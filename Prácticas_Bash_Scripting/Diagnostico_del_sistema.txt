#!/bin/bash

# Diagnostico basico de procesos Linux

usuario=$(whoami)
equipo=$(hostname)
fecha=$(date '+%Y-%m-%d %H:%M:%S')
procesos=$(ps -e --no-headers | wc -l)

echo "===== DIAGNOSTICO DEL SISTEMA ====="
echo "Usuario: $usuario"
echo "Equipo: $equipo"
echo "Fecha: $fecha"
echo "Procesos activos: $procesos"

if [ "$procesos" -gt 100 ]; then
    echo "Aviso: hay mas de 100 procesos."
else
    echo "Numero de procesos dentro del umbral definido."
fi
