#!/bin/bash

# Monitorizacion del espacio en disco

umbral_disco=80

echo "======================================"
echo "       MONITOR DE DISCO"
echo "======================================"
echo "Servidor: $(hostname)"
echo "Fecha: $(date '+%Y-%m-%d %H:%M:%S')"
echo

echo "===== ESTADO DE LOS DISCOS ====="
df -hT
echo

echo "===== COMPROBACION DE ESPACIO ====="

df -P -x tmpfs -x devtmpfs | awk -v umbral="$umbral_disco" '
NR == 1 {
    print
    next
}
{
    uso = $5
    gsub(/%/, "", uso)

    if (uso >= umbral) {
        print "AVISO: espacio ocupado igual o superior al " umbral "%."
        print "Particion: " $6 " | Uso: " uso "%"
        encontrados = 1
    }
}
END {
    if (!encontrados) {
        print "Todas las particiones comprobadas estan por debajo del umbral."
    }
}'

echo
echo "===== MONITORIZACION FINALIZADA ====="

exit 0
