
#!/bin/bash

# Monitorizacion basica de CPU y memoria RAM

umbral_ram=80
umbral_cpu=80

echo "===== MONITOR DE RECURSOS ====="
echo "Fecha: $(date '+%Y-%m-%d %H:%M:%S')"
echo

# Obtener porcentaje de memoria RAM utilizada
ram_usada=$(free | awk '/Mem:/ {printf "%.0f", $3/$2*100}')

# Obtener porcentaje de CPU inactiva
cpu_libre=$(LC_ALL=C top -bn1 | awk -F'[, ]+' '
/Cpu\(s\)/ {
    for (i=1; i<=NF; i++) {
        if ($i=="id") {
            print $(i-1)
            exit
        }
    }
}')

# Calcular porcentaje de CPU utilizada
cpu_usada=$(awk -v libre="$cpu_libre" \
    'BEGIN {printf "%.0f", 100-libre}')

echo "RAM utilizada: ${ram_usada}%"
echo "CPU utilizada: ${cpu_usada}%"
echo

# Comprobar umbral de memoria RAM
if [ "$ram_usada" -ge "$umbral_ram" ]; then
    echo "AVISO: uso de RAM igual o superior al ${umbral_ram}%."
else
    echo "RAM dentro del umbral definido."
fi

# Comprobar umbral de CPU
if [ "$cpu_usada" -ge "$umbral_cpu" ]; then
    echo "AVISO: uso de CPU igual o superior al ${umbral_cpu}%."
else
    echo "CPU dentro del umbral definido."
fi

exit 0
