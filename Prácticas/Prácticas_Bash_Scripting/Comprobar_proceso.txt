
#!/bin/bash

# Comprobar si un proceso esta activo

if [ "$#" -ne 1 ]; then
    echo "Uso: bash comprobar_proceso.sh NOMBRE_PROCESO"
    exit 2
fi

proceso="$1"

if [[ ! "$proceso" =~ ^[a-zA-Z0-9_.-]+$ ]]; then
    echo "Error: nombre de proceso no valido."
    exit 2
fi

echo "===== COMPROBACION DE PROCESO ====="
echo "Proceso buscado: $proceso"

if pgrep -x -- "$proceso" > /dev/null; then
    echo "Estado: ACTIVO"
    echo "Informacion del proceso:"
    pgrep -a -x -- "$proceso"
else
    echo "Estado: NO ENCONTRADO"
    exit 1
fi

exit 0
